# needs them. Enable with:  INTERCHANGE_TRACING=1  (and a local Phoenix running).

opentelemetry-sdk                   # tracer + span processors
opentelemetry-exporter-otlp-proto-http  # OTLP/HTTP exporter to the local Phoenix
openinference-semantic-conventions  # LLM/RAG span-kind attribute vocabulary
